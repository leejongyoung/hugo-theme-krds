{{- /* 위키 설정(params.krds.wiki). repo(예: owner/name)가 있으면 GitHub 편집·변경 이력·새 문서 주소를 만든다. repo 가 없으면 해당 링크는 나오지 않는다. 키: repo, branch(기본 main), contentDir(기본 content), section(기본 wiki) 돌려주는 값(dict): base(저장소 주소), branch, dir(콘텐츠 폴더), newBase(새 문서 만들기 주소 앞부분) */ -}} {{- $krds := site.Params.krds | default dict -}} {{- $cfg := $krds.wiki | default dict -}} {{- $repo := $cfg.repo | default "" -}} {{- $branch := $cfg.branch | default "main" -}} {{- $dir := strings.Trim ($cfg.contentdir | default "content") "/" -}} {{- $section := $cfg.section | default "wiki" -}} {{- $out := dict "base" "" "branch" $branch "dir" $dir "newBase" "" -}} {{- if $repo -}} {{- $base := printf "https://github.com/%s" $repo -}} {{- $out = dict "base" $base "branch" $branch "dir" $dir "newBase" (printf "%s/new/%s/%s/%s?filename=" $base $branch $dir $section) -}} {{- end -}} {{- return $out -}}
